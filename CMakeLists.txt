cmake_minimum_required(VERSION 3.22)

project(DashBoard_re_1010 C ASM)

set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS ON)

set(MCU_FLAGS
    -mcpu=cortex-m7
    -mthumb
    -mfpu=fpv5-d16
    -mfloat-abi=hard
)

set(COMMON_FLAGS
    -Wall
    -fdata-sections
    -ffunction-sections
)

set(LINKER_SCRIPT ${CMAKE_SOURCE_DIR}/STM32H743IITX_FLASH.ld)

#
# Directory convention (same as LP02_VCU):
#   Core/Inc, Core/Src   CubeMX-generated only, not hand-edited
#   Core/User            all hand-written application code, .c and .h together
#   Core/User/ttr_can    CAN codec generated from the DBC, not hand-edited
#   Core/User/ui         screens generated by EEZ Studio, not hand-edited
#
file(GLOB_RECURSE APP_SOURCES CONFIGURE_DEPENDS
    Core/Src/*.c
    Core/User/*.c
    Core/User/ttr_can/*.c
    Core/User/ui/*.c
)

# HAL sources are globbed rather than listed one by one as in LP02_VCU. This
# project's HAL folder is exactly what CubeMX produced, and every .c is guarded
# by HAL_xxx_MODULE_ENABLED, so disabled modules compile to empty objects.
# Globbing means adding a peripheral in CubeMX does not require editing this.
file(GLOB STM32_HAL_SOURCES CONFIGURE_DEPENDS
    Drivers/STM32H7xx_HAL_Driver/Src/*.c
)

# Only lvgl/src. The submodule also carries tests/, examples/ and demos/, which
# target desktop builds and include via <lvgl/lvgl.h>; pulling them in does not
# compile. Failing to exclude them is why the CubeIDE Release configuration
# never built.
file(GLOB_RECURSE LVGL_SOURCES CONFIGURE_DEPENDS
    Drivers/lvgl/src/*.c
)

add_executable(${PROJECT_NAME}.elf
    ${APP_SOURCES}
    ${STM32_HAL_SOURCES}
    ${LVGL_SOURCES}
    Core/Startup/startup_stm32h743iitx.s
)

target_include_directories(${PROJECT_NAME}.elf PRIVATE
    Core/Inc
    Core/User
    Core/User/ttr_can
    Core/User/ui
    Drivers                                      # lv_conf.h lives at this level
    Drivers/lvgl
    Drivers/STM32H7xx_HAL_Driver/Inc
    Drivers/STM32H7xx_HAL_Driver/Inc/Legacy
    Drivers/CMSIS/Device/ST/STM32H7xx/Include
    Drivers/CMSIS/Include
)

target_compile_definitions(${PROJECT_NAME}.elf PRIVATE
    USE_HAL_DRIVER
    STM32H743xx
    USE_PWR_LDO_SUPPLY
    LV_CONF_INCLUDE_SIMPLE
    LV_LVGL_H_INCLUDE_SIMPLE
    $<$<CONFIG:Debug>:DEBUG>
)

# Debug uses -O2, unlike LP02_VCU's -Og.
#
# This project's dominant workload is LVGL software rendering, which is
# extremely sensitive to optimisation level: at -O0 the frame rate sat in the
# single to low double digits. The .launch file also flashes Debug, so Debug is
# what runs in the car - speed wins here. -O2 additionally saved about 143 KB
# of flash.
#
# Switch the line below to -Og temporarily when single-stepping; -g3 applies
# either way.
target_compile_options(${PROJECT_NAME}.elf PRIVATE
    ${MCU_FLAGS}
    ${COMMON_FLAGS}
    $<$<CONFIG:Debug>:-O2 -g3>
    $<$<CONFIG:Release>:-O2 -g0>
)

target_link_options(${PROJECT_NAME}.elf PRIVATE
    ${MCU_FLAGS}
    -T${LINKER_SCRIPT}
    -specs=nano.specs
    -specs=nosys.specs
    # newlib-nano omits floating point printf by default. ui_bind.c formats
    # voltages with "%.1f"; without this every such value renders blank or as
    # garbage.
    -u _printf_float
    -Wl,-Map=${CMAKE_BINARY_DIR}/${PROJECT_NAME}.map
    -Wl,--gc-sections
    -Wl,--print-memory-usage
)

target_link_libraries(${PROJECT_NAME}.elf PRIVATE c m)

add_custom_command(TARGET ${PROJECT_NAME}.elf POST_BUILD
    COMMAND ${CMAKE_SIZE} $<TARGET_FILE:${PROJECT_NAME}.elf>
    COMMAND ${CMAKE_OBJCOPY} -O ihex $<TARGET_FILE:${PROJECT_NAME}.elf> ${CMAKE_BINARY_DIR}/${PROJECT_NAME}.hex
    COMMAND ${CMAKE_OBJCOPY} -O binary $<TARGET_FILE:${PROJECT_NAME}.elf> ${CMAKE_BINARY_DIR}/${PROJECT_NAME}.bin
)
